attributes {
	name = "MyApp",
	outFile = "MyApp-setup.exe",
	compressor = "lzma",

installer {

	page.welcome {},
	page.directory {},
	page.instFiles {},
	page.finish {},

	section("Core", function()
		setOutPath(INSTDIR)
		file("myapp.exe")
		writeUninstaller(INSTDIR .. "/uninstall.exe")
	end),
}

uninstaller {
	page.confirm {},
	page.instFiles {},

	section("Uninstall", function()
		delete(INSTDIR .. "/myapp.exe")

		rmDir(INSTDIR)
	end),
}
